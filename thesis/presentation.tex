% --- SLIDE 5: MODEL ---
\begin{frame}{GARCH(1,1) Specification}
\begin{equation}
\sigma_t^2 = \omega + \alpha \varepsilon_{t-1}^2 + \beta \sigma_{t-1}^2
\end{equation}
\vspace{3mm}
\begin{center}
\begin{tabular}{ll}
\toprule
\textbf{Parameter} & \textbf{Definition} \\
\midrule
$\omega$ & Long-run variance \\
$\alpha$ & Reaction to past shocks \\
$\beta$ & Persistence of volatility \\
\bottomrule
\end{tabular}
\end{center}
\vspace{2mm}
\textcolor{accent}{Constraint:} $\alpha + \beta < 1$ for stationarity
\end{frame}
